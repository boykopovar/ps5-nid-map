cmake_minimum_required(VERSION 3.20)
project(Ps5NidMap LANGUAGES CXX)

file(GLOB moduleFiles CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/data/Modules/*.inc")
list(SORT moduleFiles)
set(moduleIncludes "")
foreach(moduleFile IN LISTS moduleFiles)
    get_filename_component(moduleName "${moduleFile}" NAME_WLE)
    if(NOT moduleName MATCHES "^[A-Za-z0-9_.+-]+$")
        message(FATAL_ERROR "Invalid module filename: ${moduleFile}")
    endif()
    string(APPEND moduleIncludes "{\nModuleBuilder moduleBuilder(mappings, \"${moduleName}\");\n#include \"${moduleFile}\"\n}\n")
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/ModuleIncludes.inc" CONTENT "${moduleIncludes}")

add_library(Ps5NidMap STATIC src/Ps5NidMap/ModuleNames.cpp src/Ps5NidMap/ExportLibraries.cpp src/Ps5NidMap/Data/Modules.cpp)
add_library(Ps5NidMap::Ps5NidMap ALIAS Ps5NidMap)
target_compile_features(Ps5NidMap PUBLIC cxx_std_20)
set_target_properties(Ps5NidMap PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(Ps5NidMap PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(MINGW)
    target_compile_options(Ps5NidMap PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
