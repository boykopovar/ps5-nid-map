cmake_minimum_required(VERSION 3.20)
project(Ps5NidMap LANGUAGES CXX)

file(GLOB moduleFiles CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/data/Modules/*.inc")
list(SORT moduleFiles)
list(LENGTH moduleFiles moduleCount)
set(moduleSources "")
if(moduleCount LESS 1)
    message(FATAL_ERROR "No module mapping files")
endif()
math(EXPR lastModule "${moduleCount} - 1")
foreach(moduleIndex RANGE ${lastModule})
    list(APPEND moduleSources "${CMAKE_CURRENT_BINARY_DIR}/generated/Module${moduleIndex}.cpp")
endforeach()
set(moduleRegistry "${CMAKE_CURRENT_BINARY_DIR}/generated/ModuleRegistry.inc")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/mapping.stamp"
    BYPRODUCTS ${moduleSources} "${moduleRegistry}"
    COMMAND "${CMAKE_COMMAND}" "-DDATA_DIRECTORY=${CMAKE_CURRENT_SOURCE_DIR}/data/Modules" "-DOUTPUT_DIRECTORY=${CMAKE_CURRENT_BINARY_DIR}/generated" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/GenerateSources.cmake"
    DEPENDS ${moduleFiles} cmake/GenerateSources.cmake
    VERBATIM)
add_custom_target(Ps5NidMapSources DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/generated/mapping.stamp")

add_library(Ps5NidMap STATIC src/Ps5NidMap/ModuleNames.cpp src/Ps5NidMap/ExportLibraries.cpp src/Ps5NidMap/Data/Modules.cpp ${moduleSources})
add_library(Ps5NidMap::Ps5NidMap ALIAS Ps5NidMap)
add_dependencies(Ps5NidMap Ps5NidMapSources)
target_compile_features(Ps5NidMap PUBLIC cxx_std_20)
set_target_properties(Ps5NidMap PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(Ps5NidMap PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(MINGW)
    target_compile_options(Ps5NidMap PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    add_executable(Ps5NidMapBenchmark main.cpp)
    target_link_libraries(Ps5NidMapBenchmark PRIVATE Ps5NidMap::Ps5NidMap)
    set_target_properties(Ps5NidMapBenchmark PROPERTIES CXX_EXTENSIONS OFF)
    if(MINGW)
        target_compile_options(Ps5NidMapBenchmark PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endif()
